cmake_minimum_required(VERSION 3.31)
project(DirectPeGuest LANGUAGES C CXX)

if(NOT EMSCRIPTEN)
    message(FATAL_ERROR "Configure using the selected product's Emscripten toolchain.")
endif()
if(NOT EXISTS "${DIRECT_PE_NATIVE_PROFILE}" OR NOT EXISTS "${DIRECT_PE_GUEST_INPUTS}")
    message(FATAL_ERROR "Real native product profile and final matching managed closure are required.")
endif()
include("${DIRECT_PE_NATIVE_PROFILE}")
include("${DIRECT_PE_GUEST_INPUTS}")
if(NOT DIRECT_PE_RUNTIME_EXCEPTION_MODEL STREQUAL "emscripten" OR NOT DIRECT_PE_INTERPRETER_ONLY OR DIRECT_PE_READY_TO_RUN)
    message(FATAL_ERROR "This fresh fixture requires the actual interpreter-only JS-EH product.")
endif()

if(NOT DIRECT_PE_RUNTIME_SINGLE_THREADED)
    message(FATAL_ERROR "This fixture's owner-thread shutdown requires a source-provenanced single-threaded CLR product.")
endif()
if(NOT DIRECT_PE_FINAL_NATIVE_PRODUCT_CLOSURE)
    message(FATAL_ERROR "Preparation metadata is not a final native-product closure. Regenerate from its matching CoreLib/framework.")
endif()
file(SHA256 "${DIRECT_PE_CORELIB}" actual_corelib_hash)
if(NOT actual_corelib_hash STREQUAL DIRECT_PE_NATIVE_CORELIB_SHA256)
    message(FATAL_ERROR "The managed closure does not match the selected native product's CoreLib.")
endif()
include("${CMAKE_CURRENT_SOURCE_DIR}/verify-stage.cmake")
foreach(input IN LISTS DIRECT_PE_RUNTIME_LIBRARY_FILES DIRECT_PE_BROWSER_LIBRARIES)
    if(NOT EXISTS "${input}")
        message(FATAL_ERROR "Missing actual product input: ${input}")
    endif()
endforeach()
if(NOT DIRECT_PE_RUNTIME_LINK_INPUTS OR NOT DIRECT_PE_RUNTIME_LIBRARY_FILES OR NOT DIRECT_PE_BROWSER_LIBRARIES)
    message(FATAL_ERROR "Empty native library graph or production JS inputs.")
endif()
foreach(input IN LISTS DIRECT_PE_RUNTIME_LINK_INPUTS)
    if(input MATCHES "corerun_static|coreclr_gen_static")
        message(FATAL_ERROR "The guest owns its host and complete application helper tables: ${input}")
    endif()
endforeach()

add_library(direct_pe_helpers OBJECT ${DIRECT_PE_HELPER_SOURCES})
target_include_directories(direct_pe_helpers PRIVATE ${DIRECT_PE_HELPER_INCLUDE_DIRS})
target_compile_definitions(direct_pe_helpers PRIVATE ${DIRECT_PE_HELPER_DEFINITIONS})
target_compile_options(direct_pe_helpers PRIVATE ${DIRECT_PE_HELPER_COMPILE_OPTIONS})

add_executable(direct_pe_guest host.cpp $<TARGET_OBJECTS:direct_pe_helpers>)
target_compile_features(direct_pe_guest PRIVATE cxx_std_17)
target_include_directories(direct_pe_guest PRIVATE ${DIRECT_PE_HOST_INCLUDE_DIRS} "${DIRECT_PE_GENERATED_DIRECTORY}")
target_compile_options(direct_pe_guest PRIVATE ${DIRECT_PE_HOST_COMPILE_OPTIONS})
target_link_libraries(direct_pe_guest PRIVATE ${DIRECT_PE_RUNTIME_LINK_INPUTS})
target_link_options(direct_pe_guest PRIVATE ${DIRECT_PE_HOST_LINK_OPTIONS}
    "SHELL:--pre-js \"${CMAKE_CURRENT_SOURCE_DIR}/bootstrap.js\""
    "SHELL:--preload-file \"${DIRECT_PE_STAGE_DIRECTORY}@/managed\""
    "SHELL:--js-library \"${CMAKE_CURRENT_SOURCE_DIR}/shutdown-library.js\""
    "-sEXPORTED_FUNCTIONS=['_main','_malloc','_free','_exit','_BrowserHost_ShutdownDotnet']")
foreach(library IN LISTS DIRECT_PE_BROWSER_LIBRARIES)
    target_link_options(direct_pe_guest PRIVATE "SHELL:--js-library \"${library}\"")
endforeach()
set_target_properties(direct_pe_guest PROPERTIES SUFFIX ".js")
